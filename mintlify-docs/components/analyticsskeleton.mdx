/** SkeletonTopItem - Skeleton for ranked list items Matches the TopItem component layout /
function SkeletonTopItem({ index }: { index: number }) {
  // Vary the progress bar width for visual variety
  const progressWidths = [100, 84, 58, 46, 36];
  const labelWidths = [120, 150, 130, 100, 90];

  return (
    <div
      className="flex items-center gap-3 py-3 border-b last:border-b-0"
      style={{ borderColor: "var(--border-primary)" }}
    >
      {/* Rank number */}
      <Skeleton
        style={{
          width: 24,
          height: 24,
          borderRadius: 4,
          flexShrink: 0,
        }}
      />
      <div className="flex-1 min-w-0">
        {/* Label */}
        <Skeleton
          style={{
            width: labelWidths[index % labelWidths.length],
            height: 16,
            marginBottom: 6,
          }}
        />
        {/* Progress bar */}
        <div
          className="rounded-full overflow-hidden"
          style={{
            height: 4,
            background: "var(--bg-tertiary)",
          }}
        >
          <Skeleton
            style={{
              width: `${progressWidths[index % progressWidths.length]}%`,
              height: 4,
              borderRadius: 9999,
            }}
          />
        </div>
      </div>
      {/* Value */}
      <Skeleton style={{ width: 35, height: 16, flexShrink: 0 }} />
    </div>
  );
}

}
