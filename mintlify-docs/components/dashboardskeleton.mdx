      />
      <div className="flex items-end justify-between gap-2">

    </SkeletonCard>
  );
}

/** SkeletonStatItem - Skeleton for stat list items (Call Performance, Bookings by Type) Matches the StatItem component layout /
function SkeletonStatItem({ showProgress = true }: { showProgress?: boolean }) {
  return (
    <div
      className="flex items-center gap-3 py-3 border-b last:border-b-0"
      style={{ borderColor: "var(--border-primary)" }}
    >
      {/* Icon box */}
      <Skeleton style={{ width: 32, height: 32, borderRadius: 4, flexShrink: 0 }} />
      <div className="flex-1 min-w-0">
        {/* Label */}
        <Skeleton style={{ width: 90, height: 16, marginBottom: showProgress ? 6 : 0 }} />
        {/* Progress bar */}
        {showProgress && (
          <Skeleton style={{ width: "60%", height: 4, borderRadius: 9999 }} />
        )}
      </div>
      {/* Value */}
      <Skeleton style={{ width: 35, height: 16, flexShrink: 0 }} />
    </div>
  );
}
