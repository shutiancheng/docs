function SkeletonTableRow({ variant = "default" }: SkeletonTableRowProps) {
  return (
    <tr
      className="border-b"
      style={{ borderColor: "var(--border-primary)" }}
    >
      {/* Customer column - avatar, name, email */}
      <td style={{ padding: "var(--space-3) var(--space-4)" }}>
        <div className="flex items-center gap-3">
          {/* Avatar */}
          <Skeleton
            variant="circle"
            style={{ width: 36, height: 36, flexShrink: 0 }}
          />
          <div>
            <div className="flex items-center gap-2">
              {/* Name */}
              <Skeleton style={{ width: variant === "commercial" ? 110 : 130, height: 16 }} />
              {/* Commercial badge */}
              {variant === "commercial" && (
                <Skeleton style={{ width: 65, height: 18, borderRadius: 4 }} />
              )}
            </div>
            {/* Company name (commercial only) */}
            {variant === "commercial" && (
              <Skeleton style={{ width: 100, height: 14, marginTop: 4 }} />
            )}
            {/* Email */}
            <Skeleton style={{ width: 160, height: 14, marginTop: 4 }} />
          </div>
        </div>
      </td>

      {/* Phone column */}
      <td style={{ padding: "var(--space-3) var(--space-4)" }}>
        <Skeleton style={{ width: 110, height: 16 }} />
      </td>

      {/* Address column */}
      <td style={{ padding: "var(--space-3) var(--space-4)" }}>
        <div>
          <Skeleton style={{ width: 140, height: 16, marginBottom: 4 }} />
          <Skeleton style={{ width: 60, height: 14 }} />
        </div>
      </td>

      {/* Bookings column */}
      <td
        className="text-center"
        style={{ padding: "var(--space-3) var(--space-4)" }}

      {/* Last Contact column */}
      <td style={{ padding: "var(--space-3) var(--space-4)" }}>
        <Skeleton style={{ width: 80, height: 16 }} />

      {/* Status column */}
      <td style={{ padding: "var(--space-3) var(--space-4)" }}>
        <Skeleton style={{ width: 55, height: 24, borderRadius: 9999 }} />
      </td>

      {/* Menu column */}
      <td style={{ padding: "var(--space-3) var(--space-4)", width: 40 }}>
        <Skeleton style={{ width: 24, height: 24 }} />
      </td>
    </tr>
  );
}
