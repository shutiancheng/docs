    </div>
  );
}

/** SkeletonTabsPanel - Skeleton for the tabs panel with content /
function SkeletonTabsPanel() {
  return (
    <SkeletonCard style={{ overflow: "hidden" }}>
      {/* Tab navigation */}
      <div
        className="flex border-b"
        style={{ borderColor: "var(--border-primary)" }}
      >
        <Skeleton style={{ width: 80, height: 44, margin: "0 4px" }} />
        <Skeleton style={{ width: 80, height: 44, margin: "0 4px" }} />
        <Skeleton style={{ width: 90, height: 44, margin: "0 4px" }} />
      </div>

      {/* Tab content - Transcript view */}
      <div className="p-4">
        {/* Search bar */}
        <Skeleton
          style={{
            width: "100%",
            height: 40,
            borderRadius: 8,
            marginBottom: 16,
          }}
        />

        {/* Transcript entries */}
        {Array.from({ length: 5 }).map((_, i) => (
          <div key={i} className="flex gap-3 p-3 mb-1">
            {/* Timestamp */}
            <Skeleton style={{ width: 40, height: 14, flexShrink: 0 }} />
            <div className="flex-1">
              {/* Speaker label */}
              <div className="flex items-center gap-2 mb-1">
                <Skeleton style={{ width: 14, height: 14 }} />
                <Skeleton style={{ width: 60, height: 12 }} />
              </div>
              {/* Message text */}
              <Skeleton style={{ width: i % 2 === 0 ? "100%" : "85%", height: 14, marginBottom: 4 }} />
              {i % 3 === 0 && <Skeleton style={{ width: "60%", height: 14 }} />}
            </div>
          </div>
        ))}
      </div>
    </SkeletonCard>
  );
}
