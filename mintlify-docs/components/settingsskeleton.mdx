/** SkeletonSection - Settings section skeleton Matches SettingsSection component exactly /
interface SkeletonSectionProps {
  rows?: number;
  titleWidth?: string;
  descriptionWidth?: string;
  includeProfilePhoto?: boolean;
}

function SkeletonSection({
  rows = 4,
  titleWidth = "160px",
  descriptionWidth = "240px",
  includeProfilePhoto = false,
}: SkeletonSectionProps) {
  // Fixed widths for consistent appearance (no random)
  const rowConfigs = [
    { label: "100px", desc: "140px", action: true },
    { label: "90px", desc: "120px", action: false },
    { label: "95px", desc: "180px", action: true },
    { label: "70px", desc: "200px", action: true },
  ];

  return (
    <SkeletonCard style={{ marginBottom: "var(--space-5)" }}>
      {/* Section header - matches SettingsSection */}
      <div
        className="border-b"
        style={{
          padding: "var(--space-4) var(--space-5)",
          borderColor: "var(--border-primary)",
        }}
      >
        {/* text-base (14px) font-medium with line-height */}
        <Skeleton
          style={{ width: titleWidth, height: 18, marginBottom: "var(--space-1)" }}
        />
        {/* text-sm (13px) with line-height */}
        <Skeleton
          style={{ width: descriptionWidth, height: 16, marginTop: "var(--space-1)" }}
        />
      </div>

      {/* Section content */}
      <div style={{ padding: "var(--space-4) var(--space-5)" }}>
        {/* Profile photo skeleton (for Account section) */}
        {includeProfilePhoto && <SkeletonProfilePhoto />}

        {Array.from({ length: rows }).map((_, i) => {
          const config = rowConfigs[i % rowConfigs.length];
          return (
            <SkeletonRow
              key={i}
              labelWidth={config.label}
              descriptionWidth={config.desc}
              hasAction={config.action}
            />
          );
        })}
      </div>
    </SkeletonCard>
  );
}
