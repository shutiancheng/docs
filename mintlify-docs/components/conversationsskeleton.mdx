/** SkeletonConversationCard - Skeleton for a single conversation card Matches the ConversationCard component layout /
interface SkeletonConversationCardProps {
  variant?: "default" | "urgent" | "booking";
}

function SkeletonConversationCard({ variant = "default" }: SkeletonConversationCardProps) {
  // Vary widths slightly for visual diversity
  const summaryWidths = {
    default: ["100%", "85%"],
    urgent: ["95%", "75%"],
    booking: ["100%", "90%"],
  };

  return (
    <SkeletonCard style={{ padding: "var(--space-4)" }}>
      {/* Header row */}
      <div className="flex items-start justify-between">
        <div className="flex items-center gap-3">
          {/* Avatar circle */}
          <Skeleton
            variant="circle"
            style={{ width: 40, height: 40, flexShrink: 0 }}
          />
          <div>
            {/* Caller name */}
            <Skeleton style={{ width: 130, height: 16, marginBottom: 4 }} />
            {/* Phone and postcode */}
            <Skeleton style={{ width: 160, height: 14 }} />
          </div>
        </div>
        <div className="text-right">
          {/* Date */}
          <Skeleton style={{ width: 50, height: 14, marginLeft: "auto" }} />
          {/* Time */}
          <Skeleton
            style={{
              width: 40,
              height: 14,
              marginTop: 4,
              marginLeft: "auto",
            }}
          />
        </div>
      </div>

      {/* Summary text */}
      <div className="mt-3">
        <Skeleton
          style={{
            width: summaryWidths[variant][0],
            height: 16,
            marginBottom: 6,
          }}
        />
        <Skeleton style={{ width: summaryWidths[variant][1], height: 16 }} />
      </div>

      {/* Footer row */}
      <div
        className="flex items-center justify-between mt-4"
        style={{
          paddingTop: "var(--space-3)",
          borderTop: "1px solid var(--border-primary)",
        }}
      >
        <div className="flex items-center gap-3">
          {/* Outcome badge */}
          <Skeleton
            style={{
              width: variant === "booking" ? 65 : 55,
              height: 24,
              borderRadius: 9999,
            }}
          />
          {/* Urgency badge (only for urgent) */}
          {variant === "urgent" && (
            <Skeleton
              style={{ width: 75, height: 24, borderRadius: 9999 }}
            />
          )}
          {/* Duration */}
          <Skeleton style={{ width: 45, height: 14 }} />
        </div>
        <div className="flex items-center gap-2">
          {/* Play button */}
          <Skeleton
            style={{ width: 50, height: 26, borderRadius: "var(--radius-sm)" }}
          />
          {/* Chevron */}
          <Skeleton style={{ width: 16, height: 16 }} />
        </div>
      </div>
    </SkeletonCard>
  );
}
